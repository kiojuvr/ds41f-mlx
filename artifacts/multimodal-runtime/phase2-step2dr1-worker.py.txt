def pair_digest(saved):
    import mlx.core as mx
    import numpy as np
    h = hashlib.sha256()
    for layer in saved.target:
        h.update(repr(layer.meta_state).encode())
        for value in (*layer.cache, layer.left_padding, layer.lengths):
            if value is not None:
                h.update(str((value.shape, value.dtype)).encode())
                h.update(np.asarray(value.view(mx.uint8)).tobytes())
    for ring in saved.rings:
        h.update(str((ring.offset, ring.max_size)).encode())
        h.update(np.asarray(ring.keys.view(mx.uint8)).tobytes())
    return h.hexdigest()

def check_restored(prepared):
    # Read-only normal acquisition before enqueue, never patched lookup results.
    import mlx.core as mx
    scheduler = backend.scheduler
    images = prepared.multimodal.identities()
    restored = scheduler.checkpoints.acquire(prepared.token_ids, images=images)
    if restored is None: raise RuntimeError('planned second request has no paired hit; stop')
    target, rings, tokens = restored
    hit = len(tokens)
    first, later = prepared.multimodal.spans
    assert 0 < hit < later.start and first.start+first.length <= hit
    assert tokens == prepared.token_ids[:hit]
    saved = next(s for s in scheduler.checkpoints._payloads.values() if s.tokens == tuple(tokens))
    validate_pair(target, rings, tokens)
    assert list(saved.images) == [first.identity()]
    comparisons = []
    for a, b in zip(target, saved.target, strict=True):
        assert a is not b and tuple(a.meta_state) == tuple(b.meta_state)
        for x, y in zip((*a.cache, a.left_padding, a.lengths),
                        (*b.cache, b.left_padding, b.lengths), strict=True):
            assert (x is None) == (y is None)
            if x is not None:
                assert x.shape == y.shape and x.dtype == y.dtype
                comparisons.append(mx.array_equal(x, y))
    for a, b in zip(rings, saved.rings, strict=True):
        assert a is not b and a.offset == b.offset and a.max_size == b.max_size
        comparisons.append(mx.array_equal(a.keys, b.keys))
    mx.eval(*comparisons)
    assert all(bool(v.item()) for v in comparisons)
    prepared.multimodal.validate_prefix(hit, list(saved.images))
    prepared.multimodal.validate_workload(hit)
    prepared.multimodal.validate_sources(hit)
    report['restore'] = dict(C=hit, exact_tokens=tokens, images=saved.images,
        target=[c.size() for c in target], rings=[r.offset for r in rings],
        exact_array_comparisons=len(comparisons), all_equal=True,
        fresh_patches=later.grid[0]*later.grid[1], fresh_pixels=later.source_pixels,
        covered_patches=first.grid[0]*first.grid[1], covered_pixels=first.source_pixels,
        fresh_absolute_span=[later.start, later.start+later.length],
        committed_state_sha256=pair_digest(saved))
    return saved

def execute(prepared, name):
    observation = report[name] = dict(token_ids=prepared.token_ids)
    backend._enqueue(prepared, tokenizer, name)
    scheduler = backend.scheduler
    request = scheduler.requests[name]
    scheduled, _ = scheduler._schedule_waiting()
    if not scheduled:
        observation['trace'] = dict(request.trace)
        raise RuntimeError(scheduler._completed[-1].error)
    owner = request.execution.owner
    def registered():
        pending = request.prompt_checkpoint
        observation.update(cached_tokens=request.cached_tokens, N=len(prepared.token_ids),
            images=request.image_identities, historical=prepared.image_is_historical,
            pending_C=len(pending.tokens), pending_target=[c.size() for c in pending.target],
            pending_rings=[r.offset for r in pending.rings],
            P5_target=owner.active_cache_offsets(), P5_rings=owner.dspark_context.offsets,
            P5_prompt=list(owner.history.prompt_tokens),
            uid=owner.uid, scheduler_batch=owner._bg is scheduler.batch_generator,
            forward_mapping=dict(scheduler.request_id_to_uid), reverse_mapping=dict(scheduler.uid_to_request_id),
            P6_plan=[request.cached_tokens, len(prepared.token_ids)-2, len(prepared.token_ids)-1])
    optional_observation(observation, 'registered', registered)
    outputs = []
    for _ in range(8):
        outputs.extend(scheduler.step().outputs)
        if not scheduler.requests: break
    if scheduler.requests or not outputs or outputs[-1].error:
        raise RuntimeError(outputs[-1].error if outputs else 'no terminal output')
    observation.update(trace=dict(request.trace), response=outputs[-1].recipe_response,
        events=[json.loads(e) for output in outputs for e in output.recipe_events],
        retired=dict(requests=list(scheduler.requests), running=list(scheduler.running),
            waiting=len(scheduler.waiting), mappings=dict(scheduler.request_id_to_uid),
            reverse_mapping=dict(scheduler.uid_to_request_id), pending_capture=request.prompt_checkpoint is not None,
            batch=scheduler.batch_generator is not None, owner=request.execution.owner is not None,
            native_closed=owner._closed, native_quiesced=owner._quiesced, native_uid=owner.uid,
            native_batch=owner._bg is not None,
            sources=len(prepared.multimodal._sources), fatal=backend.fatal_error),
        published=[])
    for saved in scheduler.checkpoints._payloads.values():
        validate_pair(saved.target, saved.rings, saved.tokens)
        observation['published'].append(dict(C=len(saved.tokens), images=saved.images,
            target=[c.size() for c in saved.target], rings=[r.offset for r in saved.rings]))
    assert request.trace['cache_published'] and request.trace['prediction_retired'] and request.trace['queue_empty']
    assert not observation.get('unavailable')
    assert observation['events'][-1]['type'] == 'message_stop'
    assert observation['P5_prompt'] == prepared.token_ids
    assert observation['pending_C'] == len(prepared.token_ids)-2
    assert observation['pending_target'] == [len(prepared.token_ids)-2]*40
    assert observation['pending_rings'] == [len(prepared.token_ids)-2]*3
    assert observation['P5_target'] == (len(prepared.token_ids),)*40
    assert observation['P5_rings'] == (len(prepared.token_ids),)*3
    assert observation['scheduler_batch']
    assert not scheduler.requests and not scheduler.running and not scheduler.waiting
    assert not scheduler.request_id_to_uid and not scheduler.uid_to_request_id
    assert scheduler.batch_generator is None and request.execution.owner is None
    assert owner._closed and owner._quiesced and owner.uid is None and owner._bg is None
    print('progress:', name, 'settled', request.trace.get('canonical_frontier'), flush=True)
    return observation

    report['cleanup'] = dict(model_weakref_dead=model_ref is None or model_ref() is None,
        active=mx.get_active_memory(), cache=mx.get_cache_memory(), peak=mx.get_peak_memory(),
        executor_shutdown=backend._executor._shutdown, profile=identity.numerical_profile())
    (OUT/'phase2-step2dr1-observation.json').write_text(json.dumps(report,indent=2,default=str)+'\n')
    print('progress: cleanup', report['cleanup'], flush=True)
